```js
// Full structure (used when backend + Supabase is connected)
{
  // Identity
  id,              // string — unique ID (UUID, auto-generated by Supabase)
  createdDate,     // ISO string — when the task was created
  updatedDate,     // ISO string — last time the task was modified

  // Core Info
  title,           // string — short task name (required)
  description,     // string — longer detail (optional)

  // Status
  completed,       // boolean — true / false
  completedDate,   // ISO string | null — when it was marked complete

  // Scheduling
  dueDate,         // ISO string | null — deadline
  reminderDate,    // ISO string | null — when to send a reminder
  isRecurring,     // boolean
  recurringRule,   // "daily" | "weekly" | "monthly" | null

  // Priority & Organization
  priority,        // "low" | "medium" | "high"
  category,        // string — e.g. "Work", "Personal"
  tags,            // string[] — e.g. ["urgent", "design"]

  // Collaboration (optional features)
  assignedTo,      // string | null — user ID or name
  comments,        // { id, author, text, date }[]

  // Tracking
  activityLog,     // { action, date, details }[]
}

// Lean version — use this for local state (Sections 4–12)
{
  id,              // string — use crypto.randomUUID()
  title,           // string (required)
  description,     // string (optional)
  completed,       // boolean
  completedDate,   // ISO string | null
  priority,        // "low" | "medium" | "high"
  category,        // string
  dueDate,         // ISO string | null
  createdDate,     // ISO string
  updatedDate,     // ISO string
}
```

```sql
id            UUID          PRIMARY KEY DEFAULT gen_random_uuid()
title         TEXT          NOT NULL
description   TEXT
completed     BOOLEAN       DEFAULT false
completed_date TIMESTAMPTZ
priority      TEXT          DEFAULT 'medium'  -- 'low' | 'medium' | 'high'
category      TEXT
due_date      TIMESTAMPTZ
```
